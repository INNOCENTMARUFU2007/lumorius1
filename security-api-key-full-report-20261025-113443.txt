function StatCard({ icon: Icon, label, value, color = 'violet' }: {
  icon: React.ElementType;
  label: string;
  value: string | number;
  color?: string;
}) {
  const colors: Record<string, string> = {
    violet:  'bg-violet-500/10  text-violet-400',
    emerald: 'bg-emerald-500/10 text-emerald-400',
    amber:   'bg-amber-500/10   text-amber-400',
    red:     'bg-red-500/10     text-red-400',
  };
  return (
    <div className="bg-gray-900 border border-gray-800 rounded-xl p-4">
      <div className={`w-9 h-9 rounded-lg flex items-center justify-center mb-3 ${colors[color]}`}>
        <Icon className="w-4.5 h-4.5" />
      </div>
      <p className="text-2xl font-bold text-white">{value}</p>
      <p className="text-sm text-gray-400 mt-0.5">{label}</p>
    </div>
  );
}

import {
  Controller, Get, Post, Delete, Body, Param, Query,
  UseGuards, Req, Ip,
} from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { ApiTags, ApiBearerAuth, ApiOperation } from '@nestjs/swagger';
import { Request } from 'express';
import { SecurityService } from './security.service';
import { CurrentUser, JwtPayload } from '../common/decorators/current-user.decorator';
import { Roles } from '../common/decorators/roles.decorator';
import { RolesGuard } from '../common/guards/roles.guard';
import { IsString, IsOptional, IsArray, IsDateString } from 'class-validator';

class CreateApiKeyDto {
  @IsString() name: string = '';
  @IsOptional() @IsArray() scopes?: string[];
  @IsOptional() @IsDateString() expiresAt?: string;
}

class AuditQueryDto {
  @IsOptional() @IsString() action?: string;
  @IsOptional() @IsString() userId?: string;
  @IsOptional() @IsString() from?: string;
  @IsOptional() @IsString() to?: string;
}

@ApiTags('security')
@ApiBearerAuth()
@UseGuards(AuthGuard('jwt'), RolesGuard)
@Controller('security')
export class SecurityController {
  constructor(private readonly securityService: SecurityService) {}

  @Get('summary')
  @ApiOperation({ summary: 'Security dashboard summary' })
  async summary(@CurrentUser() user: JwtPayload) {
    return this.securityService.getSecuritySummary(user.organizationId!);
  }

  @Get('audit-logs')
  @Roles('owner', 'admin')
  @ApiOperation({ summary: 'Get audit logs for the organisation' })
  async auditLogs(
    @CurrentUser() user: JwtPayload,
    @Query() query: AuditQueryDto,
  ) {
    return this.securityService.getAuditLogs(user.organizationId!, query);
  }

  @Post('api-keys')
  @ApiOperation({ summary: 'Create a new API key' })
  async createKey(
    @CurrentUser() user: JwtPayload,
    @Body() dto: CreateApiKeyDto,
  ) {
    return this.securityService.createApiKey(user.organizationId!, user.sub, dto);
  }

  @Get('api-keys')
  @ApiOperation({ summary: 'List all active API keys' })
  async listKeys(@CurrentUser() user: JwtPayload) {
    return this.securityService.listApiKeys(user.organizationId!);
  }

  @Delete('api-keys/:id')
  @ApiOperation({ summary: 'Revoke an API key' })
  async revokeKey(
    @CurrentUser() user: JwtPayload,
    @Param('id') keyId: string,
  ) {
    return this.securityService.revokeApiKey(user.organizationId!, keyId);
  }

  @Get('login-attempts')
  @Roles('owner', 'admin')
  @ApiOperation({ summary: 'Recent login attempts for the organisation' })
  async loginAttempts(@CurrentUser() user: JwtPayload) {
    return this.securityService.getLoginAttempts(user.organizationId!);
  }
}

import {
  Injectable,
  Logger,
  NotFoundException,
} from '@nestjs/common';
import { v4 as uuidv4 } from 'uuid';
import * as bcrypt from 'bcrypt';
import * as crypto from 'crypto';

import { DatabaseService } from '../database/database.service';
import { EmailService } from '../common/email/email.service';

export interface AuditLogEntry {
  organizationId?: string;
  userId?: string;
  action: string;
  resourceType?: string;
  resourceId?: string;
  metadata?: Record<string, unknown>;
  ipAddress?: string;
  userAgent?: string;
}

@Injectable()
export class SecurityService {
  private readonly logger = new Logger(SecurityService.name);

  constructor(
    private readonly db: DatabaseService,
    private readonly emailService: EmailService,
  ) {}

  // â”€â”€ Audit Logs â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

  async logAudit(entry: AuditLogEntry) {
    try {
      await this.db.query(
        `INSERT INTO audit_logs (
           organization_id,
           user_id,
           action,
           resource_type,
           resource_id,
           metadata,
           ip_address,
           user_agent
         )
         VALUES ($1, $2, $3, $4, $5, $6, $7, $8)`,
        [
          entry.organizationId ?? null,
          entry.userId ?? null,
          entry.action,
          entry.resourceType ?? null,
          entry.resourceId ?? null,
          JSON.stringify(entry.metadata ?? {}),
          entry.ipAddress ?? null,
          entry.userAgent ?? null,
        ],
      );
    } catch (error) {
      this.logger.error(
        'Failed to write audit log',
        error instanceof Error ? error.stack : String(error),
      );
    }
  }

  async getAuditLogs(
    organizationId: string,
    filters: {
      action?: string;
      userId?: string;
      from?: string;
      to?: string;
      limit?: number;
    },
  ) {
    const conditions: string[] = ['al.organization_id = $1'];
    const params: unknown[] = [organizationId];
    let idx = 2;

    if (filters.action) {
      conditions.push(`al.action ILIKE $${idx++}`);
      params.push(`%${filters.action}%`);
    }

    if (filters.userId) {
      conditions.push(`al.user_id = $${idx++}`);
      params.push(filters.userId);
    }

    if (filters.from) {
      conditions.push(`al.created_at >= $${idx++}`);
      params.push(filters.from);
    }

    if (filters.to) {
      conditions.push(`al.created_at <= $${idx++}`);
      params.push(filters.to);
    }

    const limit = Math.max(
      1,
      Math.min(
        Math.floor(filters.limit ?? 100),
        500,
      ),
    );

    params.push(limit);

    const res = await this.db.query(
      `SELECT
         al.*,
         u.name AS user_name,
         u.email AS user_email
       FROM audit_logs al
       LEFT JOIN users u ON al.user_id = u.id
       WHERE ${conditions.join(' AND ')}
       ORDER BY al.created_at DESC
       LIMIT $${idx}`,
      params,
    );

    return res.rows;
  }

  // â”€â”€ API Keys â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  //
  // The canonical API-key ownership column in the database
  // is created_by_id, not user_id.

  async createApiKey(
    organizationId: string,
    userId: string,
    dto: {
      name: string;
      scopes?: string[];
      expiresAt?: string;
    },
  ) {
    const rawKey = `nxs_${crypto.randomBytes(32).toString('hex')}`;
    const prefix = rawKey.substring(0, 12);
    const keyHash = await bcrypt.hash(rawKey, 10);
    const id = uuidv4();

    await this.db.query(
      `INSERT INTO api_keys (
         id,
         organization_id,
         created_by_id,
         name,
         key_hash,
         key_prefix,
         scopes,
         expires_at
       )
       VALUES ($1, $2, $3, $4, $5, $6, $7, $8)`,
      [
        id,
        organizationId,
        userId,
        dto.name,
        keyHash,
        prefix,
        dto.scopes ?? [],
        dto.expiresAt ?? null,
      ],
    );

    // Notify the user; a mail failure must not undo key creation.
    try {
      const user = await this.db.findOne<Record<string, unknown>>(
        'users',
        { id: userId },
      );

      if (user?.email) {
        void this.emailService
          .sendApiKeyCreated(
            String(user.email),
            String(user.name ?? ''),
            dto.name,
            prefix,
          )
          .catch((error: unknown) => {
            this.logger.warn(
              'Could not send API-key creation email.',
              error instanceof Error
                ? error.message
                : String(error),
            );
          });
      }
    } catch (error) {
      this.logger.warn(
        'API key was created, but its notification could not be prepared.',
        error instanceof Error
          ? error.message
          : String(error),
      );
    }

    // Return the plaintext key once only. Never persist it.
    return {
      id,
      name: dto.name,
      key: rawKey,
      prefix,
      createdAt: new Date().toISOString(),
    };
  }

  async listApiKeys(organizationId: string) {
    const res = await this.db.query(
      `SELECT
         ak.id,
         ak.name,
         ak.key_prefix,
         ak.scopes,
         ak.last_used_at,
         ak.expires_at,
         ak.is_active,
         ak.created_at,
         u.name AS created_by
       FROM api_keys ak
       JOIN users u ON ak.created_by_id = u.id
       WHERE ak.organization_id = $1
         AND ak.is_active = TRUE
       ORDER BY ak.created_at DESC`,
      [organizationId],
    );

    return res.rows;
  }

  async revokeApiKey(
    organizationId: string,
    keyId: string,
  ) {
    const result = await this.db.query(
      `UPDATE api_keys
       SET is_active = FALSE
       WHERE id = $1
         AND organization_id = $2`,
      [keyId, organizationId],
    );

    if (result.rowCount === 0) {
      throw new NotFoundException('API key not found');
    }

    return { revoked: true };
  }

  async validateApiKey(
    rawKey: string,
  ): Promise<Record<string, unknown> | null> {
    if (!rawKey.startsWith('nxs_')) {
      return null;
    }

    const prefix = rawKey.substring(0, 12);

    const candidates = await this.db.query(
      `SELECT *
       FROM api_keys
       WHERE key_prefix = $1
         AND is_active = TRUE`,
      [prefix],
    );

    for (
      const candidate of candidates.rows as Record<string, unknown>[]
    ) {
      if (
        candidate.expires_at &&
        new Date(candidate.expires_at as string) < new Date()
      ) {
        continue;
      }

      const matches = await bcrypt.compare(
        rawKey,
        String(candidate.key_hash),
      );

      if (matches) {
        await this.db.query(
          `UPDATE api_keys
           SET last_used_at = NOW()
           WHERE id = $1`,
          [candidate.id],
        );

        return candidate;
      }
    }

    return null;
  }

  // â”€â”€ Login Attempts â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

  async getLoginAttempts(
    organizationId: string,
    limit = 50,
  ) {
    const safeLimit = Math.max(
      1,
      Math.min(Math.floor(limit), 500),
    );

    const res = await this.db.query(
      `SELECT la.*
       FROM login_attempts la
       JOIN users u ON la.email = u.email
       WHERE u.organization_id = $1
       ORDER BY la.created_at DESC
       LIMIT $2`,
      [organizationId, safeLimit],
    );

    return res.rows;
  }

  // â”€â”€ Security Summary â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

  async getSecuritySummary(organizationId: string) {
    const [
      apiKeys,
      recentLogins,
      failedLogins,
      sessions,
    ] = await Promise.all([
      this.db.query(
        `SELECT COUNT(*) AS count
         FROM api_keys
         WHERE organization_id = $1
           AND is_active = TRUE`,
        [organizationId],
      ),

      this.db.query(
        `SELECT COUNT(*) AS count
         FROM login_attempts la
         JOIN users u ON la.email = u.email
         WHERE u.organization_id = $1
           AND la.success = TRUE
           AND la.created_at > NOW() - INTERVAL '24 hours'`,
        [organizationId],
      ),

      this.db.query(
        `SELECT COUNT(*) AS count
         FROM login_attempts la
         JOIN users u ON la.email = u.email
         WHERE u.organization_id = $1
           AND la.success = FALSE
           AND la.created_at > NOW() - INTERVAL '24 hours'`,
        [organizationId],
      ),

      this.db.query(
        `SELECT COUNT(*) AS count
         FROM sessions s
         JOIN users u ON s.user_id = u.id
         WHERE u.organization_id = $1
           AND s.is_valid = TRUE`,
        [organizationId],
      ),
    ]);

    const mfaAdoption = await this.db.query(
      `SELECT
         COUNT(*) FILTER (
           WHERE mfa_enabled = TRUE
         ) AS mfa_enabled_count,
         COUNT(*) AS total_count
       FROM users
       WHERE organization_id = $1
         AND is_active = TRUE`,
      [organizationId],
    );

    const row = (
      mfaAdoption.rows[0] ?? {}
    ) as Record<string, unknown>;

    const total = Number(row.total_count ?? 0);
    const mfaCount = Number(row.mfa_enabled_count ?? 0);

    const getCount = (
      rows: Array<Record<string, unknown>>,
    ) => Number(rows[0]?.count ?? 0);

    return {
      apiKeysActive: getCount(apiKeys.rows),
      loginsLast24h: getCount(recentLogins.rows),
      failedLoginsLast24h: getCount(failedLogins.rows),
      activeSessions: getCount(sessions.rows),
      mfaAdoptionRate:
        total > 0
          ? Math.round((mfaCount / total) * 100)
          : 0,
      mfaEnabledUsers: mfaCount,
      totalUsers: total,
    };
  }
}

import {
  Injectable, CanActivate, ExecutionContext, UnauthorizedException,
} from '@nestjs/common';
import { Request } from 'express';
import { SecurityService } from '../../security/security.service';

@Injectable()
export class ApiKeyGuard implements CanActivate {
  constructor(private readonly securityService: SecurityService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const req = context.switchToHttp().getRequest<Request>();
    const apiKey = this.extractKey(req);
    if (!apiKey) throw new UnauthorizedException('API key required');

    const keyRecord = await this.securityService.validateApiKey(apiKey);
    if (!keyRecord) throw new UnauthorizedException('Invalid or revoked API key');

    // Attach key metadata to request
    (req as any).apiKey = keyRecord;
    return true;
  }

  private extractKey(req: Request): string | null {
    const header = req.headers['x-api-key'] as string | undefined;
    if (header) return header;
    const bearer = req.headers['authorization'];
    if (bearer?.startsWith('ApiKey ')) return bearer.slice(7);
    return null;
  }
}

import {
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';

import { PassportStrategy } from '@nestjs/passport';
import {
  ExtractJwt,
  Strategy,
} from 'passport-jwt';

import { ConfigService } from '@nestjs/config';

import { DatabaseService } from '../../database/database.service';

import {
  JwtPayload,
} from '../../common/decorators/current-user.decorator';

@Injectable()
export class JwtStrategy extends PassportStrategy(
  Strategy,
) {
  constructor(
    private readonly configService: ConfigService,
    private readonly db: DatabaseService,
  ) {
    const secret =
      configService.get<string>('jwt.secret') ??
      configService.get<string>('JWT_SECRET') ??
      process.env.JWT_SECRET;

    if (!secret) {
      throw new Error(
        'JWT_SECRET is not configured. ' +
          'Set JWT_SECRET in the Lumorius .env file.',
      );
    }

    super({
      jwtFromRequest:
        ExtractJwt.fromAuthHeaderAsBearerToken(),

      ignoreExpiration: false,

      secretOrKey: secret,
    });
  }

  async validate(
    payload: JwtPayload,
  ): Promise<JwtPayload> {
    const user =
      await this.db.findOne(
        'users',
        {
          id: payload.sub,
          is_active: true,
        },
      );

    if (!user) {
      throw new UnauthorizedException(
        'User not found or inactive',
      );
    }
    const tokenCutoff = (user as Record<string, unknown>).tokens_valid_after;
    if (tokenCutoff) {
      const cutoffMs = new Date(String(tokenCutoff)).getTime();
      const issuedAt = (payload as JwtPayload & { iat?: number }).iat;
      const cutoffSeconds = Math.floor(cutoffMs / 1000);
      if (!Number.isFinite(cutoffMs) || typeof issuedAt !== 'number' || issuedAt <= cutoffSeconds) {
        throw new UnauthorizedException('Session revoked. Please sign in again.');
      }
    }

    return payload;
  }
}
